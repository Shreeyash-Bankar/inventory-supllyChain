async setGimbalRateHighRate(
  pitchRateDegS = 0,

Add this method to JoystickService:

handleGimbalButtons() {
  if (!this.buttons.length) return;

  const {
    pitchUp,
    pitchDown,
    yawLeft,
    yawRight,
  } = JoystickService.GIMBAL_CONFIG.buttons;

  const isPressed = (index) =>
    Number(this.buttons[index]) === 1;

  const pitchUpPressed = isPressed(pitchUp);
  const pitchDownPressed = isPressed(pitchDown);
  const yawLeftPressed = isPressed(yawLeft);
  const yawRightPressed = isPressed(yawRight);

  let pitchRate = 0;
  let yawRate = 0;

  /*
   * Pitch
   *
   * UP    = positive pitch rate
   * DOWN  = negative pitch rate
   */
  if (pitchUpPressed && !pitchDownPressed) {
    pitchRate = JoystickService.GIMBAL_CONFIG.rateDegS;
  } else if (pitchDownPressed && !pitchUpPressed) {
    pitchRate = -JoystickService.GIMBAL_CONFIG.rateDegS;
  }

  /*
   * Yaw
   *
   * RIGHT = positive yaw rate
   * LEFT  = negative yaw rate
   */
  if (yawRightPressed && !yawLeftPressed) {
    yawRate = JoystickService.GIMBAL_CONFIG.rateDegS;
  } else if (yawLeftPressed && !yawRightPressed) {
    yawRate = -JoystickService.GIMBAL_CONFIG.rateDegS;
  }

  /*
   * Detect change.
   *
   * Don't send a new command if nothing changed.
   */
  if (
    pitchRate === this.gimbal.pitchRate &&
    yawRate === this.gimbal.yawRate
  ) {
    return;
  }

  this.gimbal.pitchRate = pitchRate;
  this.gimbal.yawRate = yawRate;

  console.log(
    `[GIMBAL] D-pad rate changed: pitch=${pitchRate} deg/s yaw=${yawRate} deg/s`,
  );

  this.sendGimbalRate();
}

You specifically asked for:

another function if this one fails where in the axis are stopped by the button opposite to it is triggered, pressed

I would make this explicit.

For example:

UP  → move pitch up
DOWN → stop pitch up

and:

RIGHT → stop yaw left

But you actually get this behavior automatically with the code above.

For example, suppose:

is stuck at 1.

So the opposite button acts as a safety stop.

If you want the opposite button to explicitly send a stop command even when the original button remains stuck, I'd make it even more defensive:

handleGimbalButtons() {
  if (!this.buttons.length) return;

  const {
    pitchUp,
    pitchDown,
    yawLeft,
    yawRight,
  } = JoystickService.GIMBAL_CONFIG.buttons;

  const pressed = (index) =>
    Number(this.buttons[index]) === 1;

  const up = pressed(pitchUp);
  const down = pressed(pitchDown);
  const left = pressed(yawLeft);
  const right = pressed(yawRight);

  let pitchRate = 0;
  let yawRate = 0;

  // --------------------------------------------------
  // PITCH
  // --------------------------------------------------

  if (up && !down) {
    pitchRate = JoystickService.GIMBAL_CONFIG.rateDegS;
  } else if (down && !up) {
    pitchRate = -JoystickService.GIMBAL_CONFIG.rateDegS;
  } else {
    // Both pressed OR neither pressed = STOP
    pitchRate = 0;
  }

  // --------------------------------------------------
  // YAW
  // --------------------------------------------------

  if (right && !left) {
    yawRate = JoystickService.GIMBAL_CONFIG.rateDegS;
  } else if (left && !right) {
    yawRate = -JoystickService.GIMBAL_CONFIG.rateDegS;
  } else {
    // Both pressed OR neither pressed = STOP
    yawRate = 0;
  }

  if (
    pitchRate === this.gimbal.pitchRate &&
    yawRate === this.gimbal.yawRate
  ) {
    return;
  }

  this.gimbal.pitchRate = pitchRate;
  this.gimbal.yawRate = yawRate;

  this.sendGimbalRate();
}

This is the behavior I recommend.

async stop() {
  if (this.interval) {
    clearInterval(this.interval);
    this.interval = null;
  }

  // Stop gimbal high-rate stream
  this.stopGimbalStream();

  // Stop gimbal movement
  this.gimbal.pitchRate = 0;
  this.gimbal.yawRate = 0;

  try {
    await this.commandService.stopGimbal();

      error,
    );
  }

    yaw: 1500,

}
